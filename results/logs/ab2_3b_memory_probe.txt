# probe script
"""Memory probe (not a registered run): peak GPU memory of a prompted-Qwen step with Qwen2.5-3B-Instruct
under the registered settings (bf16, group 8, constrained decoding, temperature 1.0, max_new_tokens 112,
full 24-line history), on the local card. Produces no search result."""
import json, random, sys, time
sys.path.insert(0, "/home/soumitra/LeagueTST-e4")
import torch
from huggingface_hub import snapshot_download
from leaguetst.architect import runner
from leaguetst.architect.llm import load, make_trie, generate
from leaguetst.architect.prompt import build_messages
from leaguetst.search.replay import Table, Space

model = sys.argv[1] if len(sys.argv) > 1 else "Qwen/Qwen2.5-3B-Instruct"
path = snapshot_download(model)
print("snapshot", path, flush=True)
rows = runner.table_rows("ILI")
table = Table(rows)
space = Space(table.ids)
tok, m = load(model)
m.eval()
trie = make_trie(tok, space, False)
rng = random.Random(0)
ids = list(table.ids)
torch.cuda.reset_peak_memory_stats()
lens = []
for rep in range(6):
    hist = [(c, 0.2 + rng.random()) for c in rng.sample(ids, 40 + 20 * rep)]
    msgs = build_messages("ILI", 24, space, hist, False)
    t = time.time()
    out = generate(tok, m, msgs, 8, 1.0, trie)
    lens.append(int(out[0].shape[0]))
    print(f"rep {rep}: prompt {out[0].shape[0]} tokens, {time.time()-t:.1f}s, peak {torch.cuda.max_memory_allocated()/2**30:.2f} GiB,"
          f" reserved {torch.cuda.max_memory_reserved()/2**30:.2f} GiB", flush=True)
free, total = torch.cuda.mem_get_info()
print(json.dumps({"model": model, "snapshot": path, "max_prompt_tokens": max(lens),
                  "peak_allocated_gib": torch.cuda.max_memory_allocated() / 2**30,
                  "peak_reserved_gib": torch.cuda.max_memory_reserved() / 2**30,
                  "card_total_gib": total / 2**30, "card_free_now_gib": free / 2**30}))
